\prob If $\vec{F}(x,y) = \nabla (x^2+y^4)$, find $\int_C \vec{F} \cdot d\vec{r}$ where $C$ is the quarter of the circle $x^2+y^2=4$ in the first quadrant, oriented counterclockwise.

\vfill
\prob Compute $\int_C \vec{F} \cdot d \vec{r}$ for each of the following parts.

\bigskip
\begin{enumerate}
\item $\vec{F}(x,y) = \begin{bmatrix} 3x^2 & 4y^3\end{bmatrix}^T$ and $C$ is the top of the unit circle from $(1,0)$ to $(-1,0)$.\vfill\newpage
\item $\vec{F}(x,y) = y\sin{(xy)}\vec{i} + x\sin{(xy)}\vec{j}$ and $C$ is the parabola $y=x^2$ from $(1,1)$ to $(2,4)$.\vfill
\item $\vec{F}(x,y,z) = \begin{bmatrix} ye^{xy} & x e^{xy} & \cos{(z)}\end{bmatrix}^T$ and $C$ consists of the line from $(0,0,\pi)$ to $(1,1,\pi)$ followed by the parabola $z=\pi x^2$ in the plane $y=1$ to the point $(3,1,9\pi)$.\vfill
\end{enumerate}
